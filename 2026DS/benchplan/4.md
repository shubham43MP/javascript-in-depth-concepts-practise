A closure happens when a function remembers variables from the scope where it was created, even after that outer function has finished executing.

function createCounter() {
  let count = 0;

  return function increment() {
    count++;
    return count;
  };
}

const counter = createCounter();

counter(); // 1
counter(); // 2
counter(); // 3


// Debouncing is a great production use case of Closures

```
const debounce = (func, delay) => {
  let timerId;

  return function(...args) {
    clearTimeout(timerId);
    timerId = setTimeout(() => func(...args), delay);
  }

}

const logMessage = debounce((message) => console.log(message), 1000);


logMessage("Hel");
logMessage("Hell");
logMessage("Hello");
logMessage("Hello W");
```
